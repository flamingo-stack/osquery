/* Convert a POSIX errno to a GPG error code */
gpg_err_code_t gpg_err_code_from_errno(int err) {
    int idx = errno_to_idx(err);
    if (idx < 0) {
        return GPG_ERR_UNKNOWN_ERRNO;
    }
    return err_code_from_index[idx];
}
