## Usage Example

```c
#include <getopt.h>
#include <stdio.h>

int main(int argc, char *argv[]) {
    int opt;

    /* Short options: -v, -o <file> */
    while ((opt = getopt(argc, argv, "vo:")) != -1) {
        switch (opt) {
        case 'v':
            printf("Verbose mode\n");
            break;
        case 'o':
            printf("Output file: %s\n", optarg);
            break;
        default:
            fprintf(stderr, "Usage: %s [-v] [-o file]\n", argv[0]);
            return 1;
        }
    }
    return 0;
}
```
