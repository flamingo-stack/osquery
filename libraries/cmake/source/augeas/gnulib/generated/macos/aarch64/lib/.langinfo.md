## Usage Example

```c
#include "langinfo.h"
#include <locale.h>
#include <stdio.h>

int main(void) {
    setlocale(LC_ALL, "");

    /* Query the locale's character encoding */
    printf("Codeset: %s\n", nl_langinfo(CODESET));

    /* Query the decimal point character */
    printf("Decimal point: %s\n", nl_langinfo(DECIMAL_POINT));

    /* Query abbreviated day names */
    printf("Monday abbrev: %s\n", nl_langinfo(ABDAY_2));

    /* Query alternate (standalone) month name */
    printf("January (alt): %s\n", nl_langinfo(ALTMON_1));

    return 0;
}
```
