## Usage Example

int main(int argc, char *argv[]) {
    int opt;

    /* Short options */
    while ((opt = getopt(argc, argv, "hv")) != -1) {
        switch (opt) {
            case 'h': printf("Usage: ...\n"); break;
            case 'v': printf("Verbose mode\n"); break;
            default:  break;
        }
    }

    /* Long options with getopt_long */

        { "help",    no_argument, NULL, 'h' },
        { "verbose", no_argument, NULL, 'v' },
        { NULL, 0, NULL, 0 }
    };

    int idx;
    while ((opt = getopt_long(argc, argv, "hv", long_opts, &idx)) != -1) { /* ... */ }
    return 0;
}
