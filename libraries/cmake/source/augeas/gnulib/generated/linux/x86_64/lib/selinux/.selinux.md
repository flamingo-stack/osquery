   automatically on non-SELinux platforms via gnulib. */

int main(void) {
    /* On non-SELinux systems, always returns 0 */
    if (!is_selinux_enabled()) {
        return 0;
    }

    security_context_t ctx = NULL;
    /* Returns -1, sets errno = ENOTSUP on stub platforms */
    if (getcon(&ctx) == -1) {
        perror("getcon");
    }
    freecon(ctx);
    return 0;
}
