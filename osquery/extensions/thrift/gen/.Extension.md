## Usage Example

```cpp
// Connect to a running osquery extension socket
auto transport = std::make_shared<apache::thrift::transport::TSocket>("/tmp/osquery.em.12345");
auto protocol  = std::make_shared<apache::thrift::protocol::TBinaryProtocol>(transport);

osquery::extensions::ExtensionClient client(protocol);
transport->open();

// Health-check the extension
osquery::extensions::ExtensionStatus status;
client.ping(status);

// Invoke a plugin registered in the extension
osquery::extensions::ExtensionPluginRequest req;
osquery::extensions::ExtensionResponse resp;
client.call(resp, "table", "my_table", req);

// Gracefully stop the extension
client.shutdown();
transport->close();
```
